         Map<String, BigDecimal> result = new LinkedHashMap<>();
 
         if (symbols == null || symbols.isEmpty()) {
             return result;
         }
 
         for (String symbol : symbols) {
             if (symbol == null || symbol.isBlank()) {
                 continue;
             }
             result.put(symbol.trim().toUpperCase(), getLatestPrice(symbol));
         }

     }
